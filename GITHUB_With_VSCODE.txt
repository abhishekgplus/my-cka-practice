#Check github CLI Version
$ gh --version
# gh version 2.92.0 (2026-04-28)
# https://github.com/cli/cli/releases/tag/v2.92.0

# Log In :
gh auth login

# Step 1: Set the remote URL to the new repo
git remote set-url origin https://github.com/abhishekgplus/my-cka-practice.git

# Verify the remote
git remote -v

#Expected output:
# origin  https://github.com/abhishekgplus/my-cka-practice.git (fetch)
# origin  https://github.com/abhishekgplus/my-cka-practice.git (push)

# Create the repo
gh repo create my-cka-practice --public --source=. --remote=origin

#Push your first commit
git push -u origin main

#-u sets the upstream so future pushes can be just git push.
